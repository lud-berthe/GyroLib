cmake_minimum_required(VERSION 3.24)
project(GyroLibAudit LANGUAGES CXX)
find_package(GyroLib 1.0 CONFIG REQUIRED COMPONENTS Core)
get_target_property(audit_core_type GyroLib::gyrolib TYPE)
option(GL_AUDIT_PUBLIC_ONLY "Build public-API probes without allocation fault injection" OFF)
if(NOT GL_AUDIT_PUBLIC_ONLY AND NOT audit_core_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "These allocation-fault probes require a static core SDK.")
endif()
if(NOT GL_AUDIT_PUBLIC_ONLY)
  add_executable(sdk_audit_probes probes.cpp)
  target_link_libraries(sdk_audit_probes PRIVATE GyroLib::gyrolib)
endif()
add_executable(sdk_reaudit_probes reaudit.cpp)
target_link_libraries(sdk_reaudit_probes PRIVATE GyroLib::gyrolib)
add_executable(sdk_audit3_probes audit3.cpp)
target_link_libraries(sdk_audit3_probes PRIVATE GyroLib::gyrolib)
add_executable(sdk_audit4_probes audit4.cpp)
target_link_libraries(sdk_audit4_probes PRIVATE GyroLib::gyrolib)
enable_testing()
add_executable(sdk_audit_loop_tests ../sdk_audit_loop_tests.cpp)
target_link_libraries(sdk_audit_loop_tests PRIVATE GyroLib::gyrolib)
foreach(audit_case labels destination persistence)
  add_test(NAME installed_audit_loop_${audit_case} COMMAND sdk_audit_loop_tests ${audit_case})
endforeach()
if(MSVC AND TARGET sdk_audit_probes)
  # /EHsc assumes extern-C functions cannot throw, which is exactly the
  # contract being fault-tested. /EHa lets this probe catch the violation.
  target_compile_options(sdk_audit_probes PRIVATE /utf-8 /EHa)
endif()

add_executable(sdk_audit_f_tests audit_f.cpp)
target_link_libraries(sdk_audit_f_tests PRIVATE GyroLib::gyrolib)
foreach(scenario reconnect gap opening)
  add_test(NAME audit_f_${scenario} COMMAND sdk_audit_f_tests ${scenario})
endforeach()

option(GL_AUDIT_PANEL "Build review F frontend reproduction" OFF)
if(GL_AUDIT_PANEL)
  find_package(GyroLib 1.0 CONFIG REQUIRED COMPONENTS Core SDL Panel)
  add_executable(sdk_audit_f_panel audit_f_panel.cpp ../../third_party/imgui/backends/imgui_impl_sdlrenderer3.cpp)
  target_include_directories(sdk_audit_f_panel PRIVATE ../../third_party/imgui/backends ../../third_party/imgui)
  target_link_libraries(sdk_audit_f_panel PRIVATE GyroLib::gyrolib_sdl GyroLib::gyrolib_panel)
  add_test(NAME audit_f_open_tab COMMAND sdk_audit_f_panel)
endif()
