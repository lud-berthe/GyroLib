cmake_minimum_required(VERSION 3.24)
project(GyroLibInstallConsumer LANGUAGES C CXX)
option(GL_CONSUMER_SDL "Also exercise the installed SDL delay-load shim" OFF)
option(GL_CONSUMER_PANEL "Also check the installed panel component" OFF)
set(consumer_components Core Steam)
if(GL_CONSUMER_SDL)
    list(APPEND consumer_components SDL)
endif()
if(GL_CONSUMER_PANEL)
    list(APPEND consumer_components Panel)
endif()
find_package(GyroLib 0.2 CONFIG REQUIRED COMPONENTS ${consumer_components})
add_executable(installed_consumer main.c)
target_link_libraries(installed_consumer PRIVATE GyroLib::gyrolib)
if(GL_CONSUMER_SDL)
    target_link_libraries(installed_consumer PRIVATE GyroLib::gyrolib_sdl)
    target_compile_definitions(installed_consumer PRIVATE GL_CONSUMER_SDL)
endif()
add_executable(installed_cpp_consumer main.cpp)
target_link_libraries(installed_cpp_consumer PRIVATE GyroLib::gyrolib GyroLib::gyrolib_steam)
if(GL_CONSUMER_PANEL)
    target_link_libraries(installed_cpp_consumer PRIVATE GyroLib::gyrolib_panel)
    target_compile_definitions(installed_cpp_consumer PRIVATE GL_CONSUMER_PANEL)
endif()
enable_testing()
add_test(NAME installed_c COMMAND installed_consumer)
add_test(NAME installed_cpp COMMAND installed_cpp_consumer)
