# The demo and sensor reader have no console UI. Keep their ordinary main()
# entry points and redirected IPC streams, without allocating a Windows console.
# Command-line tests deliberately retain their console subsystem.
function(gl_windowless_executable target)
    if(WIN32 AND MSVC)
        set_target_properties(${target} PROPERTIES WIN32_EXECUTABLE TRUE)
        target_link_options(${target} PRIVATE /ENTRY:mainCRTStartup)
    endif()
endfunction()
